local function FastClick(v)
    local insetY = guiService:GetGuiInset().Y
    local posX = v.AbsolutePosition.X + (v.AbsoluteSize.X * (0.5 - v.AnchorPoint.X))
    local posY = v.AbsolutePosition.Y + (v.AbsoluteSize.Y * (0.5 - v.AnchorPoint.Y)) + insetY
 
    virtualInputManager:SendMouseButtonEvent(posX, posY, 0, true, game, 1)
    virtualInputManager:SendMouseButtonEvent(posX, posY, 0, false, game, 1)
end
 
local function setSniperState(state)
    getgenv().SniperActive = state
    if state then
        task.spawn(function()
            while getgenv().SniperActive do
                local overlay = coreGui:FindFirstChild("FoundationOverlay")
                if overlay then
                    pcall(function()
                        local elements = overlay:GetDescendants()
                        for i = 1, #elements do
                            local v = elements[i]
                            if v.Parent then
                                if v.Parent.Name == "1" and string.find(v.Name, "Buy") then
                                    local lastPos = v.AbsolutePosition
                                    task.wait(0.005)
                                    if v.AbsolutePosition == lastPos and getgenv().SniperActive then
                                        FastClick(v)
                                    end
                                elseif v.Parent.Name == "Actions" and v.Name == "1" then
                                    local isBuyBtn = false
                                    local children = v.Parent:GetDescendants()
                                    for j = 1, #children do
                                        local child = children[j]
                                        if child:IsA("TextLabel") and (string.find(child.Text, "Buy") or string.find(child.Text, "Price") or string.find(child.Text, "Robux")) then
                                            isBuyBtn = true
                                            break
                                        end
                                    end
                                    
                                    if not isBuyBtn then
                                        local lastPos = v.AbsolutePosition
                                        task.wait()
                                        if v.AbsolutePosition == lastPos and getgenv().SniperActive then
                                            FastClick(v)
                                        end
                                    end
                                end
                            end
                        end
                    end)
                end
                task.wait()
            end
        end)
    end
end

local function connect(signal, callback)
    local c = signal:Connect(callback)
    connections[#connections + 1] = c

end

local function cleanup()
    destroyed = true
    for _, c in ipairs(connections) do
        if c.Connected then
            c:Disconnect()
        end
    end
    table.clear(connections)
end

local function formatTime(seconds)
    local total = math.max(0, math.ceil(seconds))

    local days = math.floor(total / 86400)
    total %= 86400

    local hours = math.floor(total / 3600)
    total %= 3600

    local minutes = math.floor(total / 60)
    local secs = total % 60

    if days > 0 then
        return string.format("%02dd %02dh %02dm %02ds", days, hours, minutes, secs)
    elseif hours > 0 then
        return string.format("%02dh %02dm %02ds", hours, minutes, secs)
    else
        return string.format("%02dm %02ds", minutes, secs)
    end
end

local function findMarkers()
    local playerGui = player:FindFirstChildOfClass("PlayerGui")
    local markers = {}

    if not playerGui then
        return markers
    end

    for _, obj in ipairs(playerGui:GetDescendants()) do
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            local text = tostring(obj.Text or "")

            if string.find(string.lower(text), "redeemed after", 1, true) then
                markers[#markers + 1] = obj
            end
        end
    end

    return markers
end

local function findEntryData(marker)
    local current = marker

    for _ = 1, 8 do
        current = current.Parent
        if not current then
            break
        end

        local ts, fallbackName, codeMatch

        for _, child in ipairs(current:GetDescendants()) do
            if child:IsA("TextLabel") or child:IsA("TextButton") or child:IsA("TextBox") then
                local text = tostring(child.Text or "")

                if not ts then
                    local value = string.match(text, "^%s*(%d%d%d%d%d%d%d%d%d%d)%s*$")

                    if value then
                        local number = tonumber(value)

                        if number and number >= 1500000000 and number <= 2500000000 then
                            ts = number
                        end
                    end
                end

                if not codeMatch then
                    local afterCode = text:match("[Cc]ode[:;]%s*([%w_%-]+)")

                    if afterCode and afterCode ~= "" and #afterCode <= 24 then
                        codeMatch = afterCode
                    end
                end

                if not fallbackName and child ~= marker then
                    local trimmed = text:match("^%s*(.-)%s*$")

                    if trimmed ~= ""
                        and #trimmed >= 4
                        and #trimmed <= 24
                        and string.match(trimmed, "^[%u%d][%u%d_%-]*$")
                        and not BADGE_WORDS[trimmed]
                        and (not ts or trimmed ~= tostring(ts)) then
                        fallbackName = trimmed
                    end
                end
            end
        end

        if ts then
            return ts, codeMatch or fallbackName
        end
    end
end

local function loadConfig()
    if typeof(isfile) ~= "function" or typeof(readfile) ~= "function" then
        return
    end

    local ok, exists = pcall(isfile, CONFIG_PATH)
    if not ok or not exists then
        return
    end

    local decodeOk, decoded = pcall(function()
        return HttpService:JSONDecode(readfile(CONFIG_PATH))
    end)

    if not decodeOk or typeof(decoded) ~= "table" then
        return
    end

    for _, item in ipairs(decoded) do
        if item.code and item.timestamp then
            local entry = {
                code = item.code,
                timestamp = item.timestamp,
                fired = item.fired == true,
            }

            createRow(entry)
            entries[#entries + 1] = entry

            if entry.fired then
                entry.status.Text = "Redeemed"
                entry.status.TextColor3 = GREEN
            end
        end
    end
end

local function redeemEntry(entry)
    if entry.fired or destroyed then
        return
    end

    local remote = ReplicatedStorage:FindFirstChild("RedeemCode")

    if not remote or not remote:IsA("RemoteFunction") then
        entry.status.Text = "No Remote"
        entry.status.TextColor3 = Color3.fromRGB(255, 100, 100)
        return
    end

    entry.fired = true
    entry.status.Text = "Redeeming"
    entry.status.TextColor3 = Color3.fromRGB(100, 210, 255)

    local ok = pcall(function()
        remote:InvokeServer(entry.code)
    end)

    if ok then
        entry.status.Text = "Redeemed"
        entry.status.TextColor3 = GREEN
    else
        entry.fired = false
        entry.status.Text = "Failed"
        entry.status.TextColor3 = Color3.fromRGB(255, 100, 100)
    end

    saveConfig()
end

local function scanTime()
    if scanning or destroyed then
        return
    end

    scanning = true

    local markers = findMarkers()
    local added = 0
    local index = #entries

    local seen = {}
    for _, entry in ipairs(entries) do
        seen[entry.timestamp] = true
    end

    for _, marker in ipairs(markers) do
        local ts, name = findEntryData(marker)

        if ts and not seen[ts] then
            seen[ts] = true
            index += 1
            added += 1

            local entry = {
                code = name or ("Code " .. index),
                timestamp = ts,
                fired = false,
            }

            createRow(entry)
            entries[#entries + 1] = entry
        end
    end

    if #entries == 0 then
        setStatus("No codes found", Color3.fromRGB(255, 110, 110))
    elseif added > 0 then
        setStatus(#entries .. " code(s) total", GREEN)
    end

    if added > 0 then
        saveConfig()
    end

    scanning = false
end

loadConfig()

task.spawn(function()
    while not destroyed do
        scanTime()
        task.wait(SCAN_INTERVAL)
    end
end)

connect(RunService.Heartbeat, function()
    if destroyed then
        return
    end

    for _, entry in ipairs(entries) do
        if not entry.fired then
            local remaining = entry.timestamp - workspace:GetServerTimeNow()

            if remaining <= 0 then
                redeemEntry(entry)
            else
                entry.status.Text = formatTime(remaining)

                if remaining <= 3 then
                    entry.status.TextColor3 = Color3.fromRGB(255, 100, 100)
                elseif remaining <= 10 then
                    entry.status.TextColor3 = Color3.fromRGB(255, 180, 80)
                else
                    entry.status.TextColor3 = Color3.fromRGB(255, 205, 90)
                end
            end
        end
    end
end)

local dragging = false
local dragStart
local startPosition

connect(topBar.InputBegan, function(input)
    if input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseButton1 then

        dragging = true
        dragStart = input.Position
        startPosition = frame.Position

        local c
        c = input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                if c then
                    c:Disconnect()
                end
            end
        end)
    end
end)

connect(UserInputService.InputChanged, function(input)
    if not dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.Touch
        and input.UserInputType ~= Enum.UserInputType.MouseMovement then
        return
    end

    labelDelta = input.Position - dragStart

    frame.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + labelDelta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + labelDelta.Y
    )
end)

connect(close.Activated, function()
    cleanup()
    gui:Destroy()
end)
